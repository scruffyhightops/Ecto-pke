/* AUDIO - heart-monitor style scan beep */ var actx = null, beatTimer = null, beat = false; function ensureAudio(){ if(!actx){ actx = new (window.AudioContext || window.webkitAudioContext)(); } if(actx.state === 'suspended') actx.resume(); return actx; } /* one 'lub-dub' beat */ function heartBeat(){ if(offline || !settings.audio) return; var a = ensureAudio(); if(!a) return; var t = a.currentTime; var s = mode === 'HOLD' ? heldStrength : strength; var pitch = 500 + s * 300; // pitch rises with severity var vol = 0.04 + s * 0.03; // louder when severe // 'lub' - low thump beatOsc(a, t, pitch * 0.6, 0.09, vol); // 'dub' - slightly higher, follows shortly after beatOsc(a, t + 0.13, pitch * 0.8, 0.07, vol * 0.8); } function beatOsc(a, t, freq, dur, vol){ var o = a.createOscillator(), g = a.createGain(); o.type = 'sine'; o.frequency.setValueAtTime(freq, t); o.frequency.exponentialRampToValueAtTime(freq * 0.5, t + dur); g.gain.setValueAtTime(0.0001, t); g.gain.exponentialRampToValueAtTime(vol, t + 0.008); g.gain.exponentialRampToValueAtTime(0.0001, t + dur); o.connect(g).connect(a.destination); o.start(t); o.stop(t + dur + 0.02); } /* schedule beats - rate follows severity */ function startHeart(){ if(beatTimer || offline || !settings.audio) return; beat = true; beatTimer = setInterval(function(){ heartBeat(); }, 700); // base interval, adjusted below } function stopHeart(){ beat = false; if(beatTimer){ clearInterval(beatTimer); beatTimer = null; } } /* called each frame - adjusts beat rate to match strength */ function updateHeart(){ if(!beatTimer || offline || !settings.audio) return; var s = mode === 'HOLD' ? heldStrength : strength; // 700ms at calm (~85bpm) down to 430ms at severe (~140bpm) var interval = 700 - s * 270; // reschedule if the rate changed noticeably if(Math.abs(interval - beatTimer._last) > 10){ clearInterval(beatTimer); beatTimer = setInterval(function(){ heartBeat(); }, interval); beatTimer._last = interval; } }
